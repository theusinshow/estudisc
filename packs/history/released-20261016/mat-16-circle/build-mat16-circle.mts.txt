try {
  const [row] = await sql`select l.content_version, t.content_version tv from lessons l join modules m on m.id=l.module_id join tracks t on t.id=m.track_id where l.stable_id='MAT-16' and l.metadata->>'status'='published' and l.metadata->>'qaReleaseId' is not null order by l.content_version desc, t.content_version desc limit 1`;

  if (base.version !== row.content_version) throw new Error(`context v${base.version} != published v${row.content_version}`);
  const template = base.activities.filter((a: any) => a.questionId).at(-1);
  const lesson = structuredClone({ ...base, version: base.version + 1, status: "draft" });
  const fresh = authored.map(q => ({ id: `Q-MAT16-CIRCLE-0${q.n}`, version: 1, status: "draft", type: "multiple_choice", subjectCode: "MAT", stimulus: q.stimulus, stem: q.stem, items: [], assets: [],

  }
  const all = [...context.questions, ...fresh];
